#include <stdio.h>
#include <string.h>

#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

#include "esp_err.h"
#include "esp_log.h"
#include "esp_event.h"
#include "esp_netif.h"
#include "nvs_flash.h"

#include "esp_http_client.h"
#include "esp_crt_bundle.h"

#include "protocol_examples_common.h"
#include "cJSON.h"

static const char *TAG = "HEIMDALL_GPS";

static char firebase_response[1024];
static int firebase_response_len = 0;

static esp_err_t firebase_http_event_handler(esp_http_client_event_t *evt)
{
    if (evt->event_id == HTTP_EVENT_ON_DATA) {

        if (firebase_response_len + evt->data_len <
            sizeof(firebase_response) - 1) {

            memcpy(
                firebase_response + firebase_response_len,
                evt->data,
                evt->data_len
            );

            firebase_response_len += evt->data_len;

            firebase_response[firebase_response_len] = '\0';
        }
    }

    return ESP_OK;
}


static void firebase_get_location(void)
{
    firebase_response_len = 0;
    memset(firebase_response, 0, sizeof(firebase_response));

    esp_http_client_config_t config = {
        .url =
        "https://heimdall-s-watch-default-rtdb.asia-southeast1.firebasedatabase.app/live_location.json",

        .event_handler = firebase_http_event_handler,
        .crt_bundle_attach = esp_crt_bundle_attach,
        .timeout_ms = 10000
    };

    esp_http_client_handle_t client =
        esp_http_client_init(&config);

    esp_err_t err =
        esp_http_client_perform(client);

    if (err == ESP_OK) {

        ESP_LOGI(TAG, "Firebase response: %s",
                 firebase_response);

        cJSON *root =
            cJSON_Parse(firebase_response);

        if (root != NULL) {

            cJSON *latitude =
                cJSON_GetObjectItem(root, "latitude");

            cJSON *longitude =
                cJSON_GetObjectItem(root, "longitude");

            cJSON *accuracy =
                cJSON_GetObjectItem(root, "accuracy");

            if (cJSON_IsNumber(latitude) &&
                cJSON_IsNumber(longitude)) {

                printf("\n");
                printf("===== HEIMDALL GPS =====\n");

                printf(
                    "Latitude : %.6f\n",
                    latitude->valuedouble
                );

                printf(
                    "Longitude: %.6f\n",
                    longitude->valuedouble
                );

                if (cJSON_IsNumber(accuracy)) {

                    printf(
                        "Accuracy : %.1f m\n",
                        accuracy->valuedouble
                    );
                }

                printf("========================\n\n");
            }

            cJSON_Delete(root);
        }

    } else {

        ESP_LOGE(
            TAG,
            "Firebase GET failed: %s",
            esp_err_to_name(err)
        );
    }

    esp_http_client_cleanup(client);
}


void app_main(void)
{
    esp_err_t ret =
        nvs_flash_init();

    if (ret == ESP_ERR_NVS_NO_FREE_PAGES ||
        ret == ESP_ERR_NVS_NEW_VERSION_FOUND) {

        ESP_ERROR_CHECK(
            nvs_flash_erase()
        );

        ESP_ERROR_CHECK(
            nvs_flash_init()
        );
    }

    ESP_ERROR_CHECK(
        esp_netif_init()
    );

    ESP_ERROR_CHECK(
        esp_event_loop_create_default()
    );

    ESP_LOGI(
        TAG,
        "Connecting to Wi-Fi..."
    );

    ESP_ERROR_CHECK(
        example_connect()
    );

    ESP_LOGI(
        TAG,
        "Wi-Fi connected"
    );

    while (1) {

        firebase_get_location();

        vTaskDelay(
            pdMS_TO_TICKS(5000)
        );
    }
}
